   101	def parse_owned_block(fm_text):
   102	    owned = []
   103	    in_block = False
   104	    for line in fm_text.splitlines():
   105	        if re.match(r'^owned_files:\s*$', line):
   106	            in_block = True
   107	            continue
   108	        if not in_block:
   109	            continue
   110	        sm = re.match(r'^-\s+(.+?)\s*$', line)
   111	        if sm:
   112	            owned.append(sm.group(1))

   239	    """True iff any dict node in the JSON tree contains an `image_url` key."""
